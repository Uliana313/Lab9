cmake_minimum_required(VERSION 3.20)
project(Lab9 CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -pedantic)
endif()
enable_testing()

# --- Обов'язкове завдання: Library ---
add_library(library_core library/src/Library.cpp)
target_include_directories(library_core PUBLIC library/src)
add_executable(library_app library/src/main.cpp)
target_link_libraries(library_app library_core)
add_executable(library_tests tests/library_tests.cpp tests/test_main.cpp)
target_link_libraries(library_tests library_core)
target_include_directories(library_tests PRIVATE tests)
add_test(NAME library_tests COMMAND library_tests)

# --- Завдання 2: LeaderBoard (legacy і refactored проходять ті самі тести) ---
add_library(lb_legacy leaderboard/legacy/Race.cpp leaderboard/legacy/Leaderboard.cpp)
target_include_directories(lb_legacy PUBLIC leaderboard/legacy)
add_executable(lb_legacy_tests tests/leaderboard_common_tests.cpp tests/test_main.cpp)
target_link_libraries(lb_legacy_tests lb_legacy)
target_include_directories(lb_legacy_tests PRIVATE tests)
add_test(NAME lb_legacy_tests COMMAND lb_legacy_tests)

add_library(lb_refactored leaderboard/refactored/Leaderboard.cpp)
target_include_directories(lb_refactored PUBLIC leaderboard/refactored)
add_executable(lb_refactored_tests tests/leaderboard_common_tests.cpp tests/leaderboard_extension_tests.cpp tests/test_main.cpp)
target_link_libraries(lb_refactored_tests lb_refactored)
target_include_directories(lb_refactored_tests PRIVATE tests)
add_test(NAME lb_refactored_tests COMMAND lb_refactored_tests)
